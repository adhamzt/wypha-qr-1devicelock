// Otomatis mereset kunci begitu Admin mencentang checkbox atau mengetik RESET di Google Sheets!
function onEdit(e) {
  try {
    if (!e || !e.range) return;
    var sheet = e.range.getSheet();
    var sName = sheet.getName();
    var row = e.range.getRow();
    var col = e.range.getColumn();
    if (row < 2) return;

    var val = String(e.range.getValue() || '').trim().toUpperCase();
    var isResetTrigger = (val === 'TRUE' || val === 'RESET' || val === 'YA' || val === '1' || val === ' RESET SEKARANG');

    if (sName === SHEET_LOCKS && col === 9 && isResetTrigger) {
      var pid = String(sheet.getRange(row, 1).getValue() || '').trim();
      if (pid) {
        performFullResetForProject_(pid);
        e.range.setValue(false);
      }
    } else if (sName === SHEET_LEGACY_LOCKS && (col === 4 || col === 8 || col === 9) && isResetTrigger) {
      var pid2 = String(sheet.getRange(row, 1).getValue() || '').trim();
      if (pid2) {
        performFullResetForProject_(pid2);
        if (col === 9) e.range.setValue(false);
      }
    } else if (sName === SHEET_RESET_TAB && (col === 2 || col === 3) && isResetTrigger) {
      var targetPid = String(sheet.getRange(row, 1).getValue() || '').trim();
      if (targetPid) {
        if (col === 3 && (val === 'TRUE' || val === 'HAPUS_HP_2')) {
          performRemoveSecondDeviceForProject_(targetPid);
        } else {
          performFullResetForProject_(targetPid);
        }
        e.range.setValue(false);
        sheet.getRange(row, 4).setValue('✓ Direset pada ' + formatWibNow_());
      }
    } else if (sName === SHEET_CURHAT && col === 6 && val !== '') {
      sheet.getRange(row, 7).setValue(formatWibNow_());
    }
  } catch (err) {}
}

  }

}

function formatWibNow_() {
  return Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss');
}

function parseDeviceList_(raw) {
  if (!raw) return [];
  var parts = String(raw).split(/[,|;\n]+/);
  var out = [];
  for (var i = 0; i < parts.length; i++) {
    var clean = String(parts[i] || '').trim().toUpperCase();
    if (clean && out.indexOf(clean) === -1 && out.length < MAX_DEVICES) {
      out.push(clean);
    }
  }
  return out;
}

function syncProjectToLegacyAndResetSheets_(projectId, customerName, owners, statusLabel, nowStr, remoteTierVal, noteStr) {
  try {
    var ss = getSpreadsheet_();
    var legSheet = getOrCreateLegacyLockSheet(ss);
    var legData = legSheet.getDataRange().getValues();
    var target = String(projectId || '').trim().toLowerCase();
    var foundRow = -1;
    for (var i = legData.length - 1; i >= 1; i--) {
      if (String(legData[i][0] || '').trim().toLowerCase() === target) {
        foundRow = i + 1;
        break;
      }
    }
    var joinedOwners = (owners || []).join(' | ');
    if (foundRow > 0) {
      if (customerName) legSheet.getRange(foundRow, 2).setValue(customerName);
      legSheet.getRange(foundRow, 3).setValue(joinedOwners);
      legSheet.getRange(foundRow, 4).setValue(statusLabel);
      legSheet.getRange(foundRow, 6).setValue(nowStr);
      if (noteStr) legSheet.getRange(foundRow, 8).setValue(noteStr);
      if (remoteTierVal) legSheet.getRange(foundRow, 10).setValue(remoteTierVal);
    } else {
      legSheet.appendRow([
        projectId,
        customerName || projectId,
        joinedOwners,
        statusLabel,
        nowStr,
        nowStr,
        0,
        noteStr || 'Tersinkron otomatis V35',
        false,
        remoteTierVal || 'BASIC_5'
      ]);
      try { legSheet.getRange(legSheet.getLastRow(), 9).insertCheckboxes(); } catch (e) {}
    }

    var rSheet = getOrCreateResetTabSheet(ss);
    var rData = rSheet.getDataRange().getValues();
    var rRow = -1;
    for (var j = rData.length - 1; j >= 1; j--) {
      if (String(rData[j][0] || '').trim().toLowerCase() === target) {
        rRow = j + 1;
        break;
      }
    }
    if (rRow === -1) {
      rSheet.appendRow([projectId, false, false, statusLabel + ' (' + nowStr + ')', 'Centang Kolom B untuk Reset 0/2 atau Kolom C untuk Hapus HP ke-2']);
      try { rSheet.getRange(rSheet.getLastRow(), 2, 1, 2).insertCheckboxes(); } catch (e2) {}
    } else {
      rSheet.getRange(rRow, 4).setValue(statusLabel + ' (' + nowStr + ')');
    }
  } catch (err) {}
}

function performFullResetForProject_(projectId) {
  var ss = getSpreadsheet_();
  var nowStr = formatWibNow_();
  var target = String(projectId || '').trim().toLowerCase();
  try {
    PropertiesService.getScriptProperties().deleteProperty('WYPHA_LOCK_' + projectId);
  } catch (e) {}

  var lSheet = getOrCreateLockSheet(ss);
  var lData = lSheet.getDataRange().getValues();
  for (var i = 1; i < lData.length; i++) {
    if (String(lData[i][0] || '').trim().toLowerCase() === target) {
      var r = i + 1;
      lSheet.getRange(r, 3, 1, 3).setValues([['', '', 0]]);
      lSheet.getRange(r, 7).setValue(nowStr);
      lSheet.getRange(r, 9).setValue(false);
    }
  }

  var legSheet = getOrCreateLegacyLockSheet(ss);
  var legData = legSheet.getDataRange().getValues();
  for (var j = 1; j < legData.length; j++) {
    if (String(legData[j][0] || '').trim().toLowerCase() === target) {
      var lr = j + 1;
      legSheet.getRange(lr, 3).setValue('');
      legSheet.getRange(lr, 4).setValue('DIRESET (0/2 · SIAP KLAIM HP BARU)');
      legSheet.getRange(lr, 6).setValue(nowStr);
      legSheet.getRange(lr, 8).setValue('Direset penuh pada ' + nowStr);
      legSheet.getRange(lr, 9).setValue(false);
    }
  }
  SpreadsheetApp.flush();
}

function performRemoveSecondDeviceForProject_(projectId) {
  var ss = getSpreadsheet_();
  var nowStr = formatWibNow_();
  var target = String(projectId || '').trim().toLowerCase();

  var lSheet = getOrCreateLockSheet(ss);
  var lData = lSheet.getDataRange().getValues();
  var firstDev = '';
  for (var i = 1; i < lData.length; i++) {
    if (String(lData[i][0] || '').trim().toLowerCase() === target) {
      var r = i + 1;
      firstDev = String(lData[i][2] || '').trim().toUpperCase();
      lSheet.getRange(r, 4, 1, 2).setValues([['', firstDev ? 1 : 0]]);
      lSheet.getRange(r, 7).setValue(nowStr);
      lSheet.getRange(r, 9).setValue(false);
    }
  }

  var legSheet = getOrCreateLegacyLockSheet(ss);
  var legData = legSheet.getDataRange().getValues();
  for (var j = 1; j < legData.length; j++) {
    if (String(legData[j][0] || '').trim().toLowerCase() === target) {
      var lr = j + 1;
      var owners = parseDeviceList_(legData[j][2]);
      var keepFirst = firstDev || owners[0] || '';
      legSheet.getRange(lr, 3).setValue(keepFirst);
      legSheet.getRange(lr, 4).setValue(keepFirst ? 'TERKUNCI (1/2)' : 'DIRESET (0/2)');
      legSheet.getRange(lr, 6).setValue(nowStr);
      legSheet.getRange(lr, 8).setValue('HP ke-2 dihapus pada ' + nowStr + ' (Sisa Slot 1/2)');
      legSheet.getRange(lr, 9).setValue(false);
    }
  }
  try {
    if (firstDev) PropertiesService.getScriptProperties().setProperty('WYPHA_LOCK_' + projectId, firstDev);
  } catch (e) {}
  SpreadsheetApp.flush();
}

// Menu Action 2: Reset Kunci Penuh (0/2) pada Baris yang Dipilih
function resetSelectedRowLock() {
  var ss = getSpreadsheet_();
  var sheet = ss.getActiveSheet();
  var range = sheet.getActiveRange();
  if (!range || range.getRow() < 2) {
    SpreadsheetApp.getUi().alert('Pilih salah satu baris data Customer (baris ke-2 atau lebih) pada tab WyphaDeviceLocks, Daftar_Kunci_Perangkat, atau WyphaReset_Cepat terlebih dahulu.');
    return;
  }
  var row = range.getRow();
  var projId = String(sheet.getRange(row, 1).getValue() || '').trim();
  var custName = String(sheet.getRange(row, 2).getValue() || '').trim();
  if (!projId) {
    SpreadsheetApp.getUi().alert('Baris yang dipilih tidak memiliki Project ID di Kolom A.');
    return;
  }
  performFullResetForProject_(projId);
  SpreadsheetApp.getUi().alert('✓ Kunci perangkat untuk "' + (custName || projId) + '" berhasil di-reset ke 0/2! Sekarang 2 perangkat baru dapat mengklaim kunci.');
}

// Menu Action 3: Hapus HP ke-2 Saja pada Baris yang Dipilih

  var ss = getSpreadsheet_();

  var range = sheet.getActiveRange();

    SpreadsheetApp.getUi().alert('Pilih salah satu baris data Customer (baris ke-2 atau lebih) terlebih dahulu.');
    return;

    SpreadsheetApp.getUi().alert('Baris yang dipilih tidak memiliki Project ID di Kolom A.');

  SpreadsheetApp.getUi().alert('✓ HP ke-2 untuk Project ID "' + projId + '" berhasil dihapus! HP ke-1 tetap aktif (1/2) dan Slot 2/2 kini kosong.');
}

// Menu Action 6: Reset Semua Kunci Perangkat
function resetAllDeviceLocks() {
  var ui = SpreadsheetApp.getUi();
  var resp = ui.alert('Reset Semua Kunci?', 'Apakah Anda yakin ingin mereset semua kunci perangkat di tabel ini?', ui.ButtonSet.YES_NO);
  if (resp !== ui.Button.YES) return;
  try { PropertiesService.getScriptProperties().deleteAllProperties(); } catch (e) {}
  var ss = getSpreadsheet_();
  var lSheet = getOrCreateLockSheet(ss);
  if (lSheet.getLastRow() >= 2) lSheet.getRange(2, 1, lSheet.getLastRow() - 1, 10).clearContent();
  var legSheet = getOrCreateLegacyLockSheet(ss);
  if (legSheet.getLastRow() >= 2) legSheet.getRange(2, 1, legSheet.getLastRow() - 1, 10).clearContent();
  ui.alert('✓ Seluruh data kunci perangkat telah dibersihkan.');
}

function getOrCreateClientFolder(folderName, parentFolderId) {
  var rootFolder = null;
  if (parentFolderId && String(parentFolderId).trim()) {
    try { rootFolder = DriveApp.getFolderById(String(parentFolderId).trim()); } catch (e) { rootFolder = null; }
  }
  if (!rootFolder) {
    var it = DriveApp.getFoldersByName(DEFAULT_ROOT_FOLDER_NAME);
    rootFolder = it.hasNext() ? it.next() : DriveApp.createFolder(DEFAULT_ROOT_FOLDER_NAME);
  }
  var cleanSub = String(folderName || 'proyek-wypha').trim().replace(/[\\/:*?"<>|]+/g, '-') || 'proyek-wypha';
  var subIt = rootFolder.getFoldersByName(cleanSub);
  var clientFolder = subIt.hasNext() ? subIt.next() : rootFolder.createFolder(cleanSub);
  try { clientFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  return clientFolder;
}

  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var params = (e && e.parameter) ? e.parameter : {};
  // Jika diakses via GET tanpa action & tanpa deviceId (misal tes URL di browser atau 302 redirect), jalankan sync_hub agar tidak error MISSING_DEVICE_ID!
  if (!params.action && !params.deviceId) {
    params.action = 'sync_hub';
  }
  return handleRequest(params);
}

function doPost(e) {
  var body = {};
  if (e && e.parameter) {
    for (var k in e.parameter) {
      body[k] = e.parameter[k];
    }
  }
  try {
    if (e && e.postData && e.postData.contents) {
      var parsed = JSON.parse(e.postData.contents);
      if (parsed && typeof parsed === 'object') {
        for (var pk in parsed) {
          body[pk] = parsed[pk];
        }
      }
    }
  } catch (err) {}
  if (!body.action && !body.deviceId) {
    body.action = 'sync_hub';
  }
  return handleRequest(body);
}

function isTargetProjectMatch_(sheetTargetRaw, candidates) {
  var cleanTarget = String(sheetTargetRaw || '').trim().toLowerCase();
  if (!cleanTarget || cleanTarget === 'all' || cleanTarget === 'semua' || cleanTarget === '*') {
    return true;
  }
  for (var i = 0; i < candidates.length; i++) {
    var c = String(candidates[i] || '').trim().toLowerCase();
    if (!c) continue;
    if (cleanTarget === c || cleanTarget.indexOf(c) !== -1 || c.indexOf(cleanTarget) !== -1) {
      return true;
    }
  }
  return false;
}

function readActiveBroadcast(projectId, altProjectId, personName) {
  try {
    var ss = getSpreadsheet_();
    var bSheet = getOrCreateBroadcastSheet(ss);
    var rows = bSheet.getDataRange().getValues();
    var candidates = [projectId, altProjectId, personName];

    // Cari dari baris paling bawah (terbaru) ke atas
    for (var i = rows.length - 1; i >= 1; i--) {
      var activeRaw = String(rows[i][0] || '').trim().toUpperCase();
      var targetRaw = String(rows[i][1] || 'ALL').trim();
      var titleStr = String(rows[i][2] || '').trim();
      var msgStr = String(rows[i][3] || '').trim();
      if (!titleStr && !msgStr) continue;

      var isExplicitlyOff = (activeRaw === 'TIDAK' || activeRaw === 'NO' || activeRaw === 'OFF' || activeRaw === '0' || activeRaw === 'FALSE' || activeRaw === 'MATI');
      var isExplicitlyOn = (activeRaw === 'YA' || activeRaw === 'YES' || activeRaw === 'ON' || activeRaw === '1' || activeRaw === 'TRUE' || activeRaw === 'AKTIF');
      var hasSpecificProjectTarget = (targetRaw && targetRaw.toUpperCase() !== 'ALL' && targetRaw.toUpperCase() !== 'SEMUA');

      // Jika diaktifkan (YA/ON/TRUE) ATAU Admin sudah mengisi ID Proyek spesifik di Kolom B (selama tidak dimatikan OFF)
      if ((isExplicitlyOn || (hasSpecificProjectTarget && !isExplicitlyOff)) && isTargetProjectMatch_(targetRaw, candidates)) {
        return {
          active: true,
          title: titleStr || 'Pengumuman Studio Wypha',
          message: msgStr,
          ctaLabel: String(rows[i][4] || 'Info Selengkapnya').trim(),
          ctaUrl: String(rows[i][5] || '').trim()
        };
      }
    }
  } catch (e) {}
  return { active: false };
}

function readRemoteTierForProject(projectId, fallbackDefaultTier, altProjectId, personName) {
  try {
    var ss = getSpreadsheet_();
    var candidates = [projectId, altProjectId, personName].map(function(x) { return String(x || '').trim().toLowerCase(); }).filter(Boolean);

    var lSheet = getOrCreateLockSheet(ss);
    var rows = lSheet.getDataRange().getValues();
    for (var i = rows.length - 1; i >= 1; i--) {
      var rowPid = String(rows[i][0] || '').trim().toLowerCase();
      if (rowPid && candidates.indexOf(rowPid) !== -1) {
        var val = String(rows[i][9] || '').trim();
        if (val) return val;
      }
    }

    var legSheet = getOrCreateLegacyLockSheet(ss);
    var legRows = legSheet.getDataRange().getValues();
    for (var j = legRows.length - 1; j >= 1; j--) {
      var legPid = String(legRows[j][0] || '').trim().toLowerCase();
      if (legPid && candidates.indexOf(legPid) !== -1) {
        var val2 = String(legRows[j][9] || '').trim();
        if (val2) return val2;
      }
    }
  } catch (e) {}
  return String(fallbackDefaultTier || '').trim();
}

function upsertRemoteTierForProject(projectId, tierCode, customerName, nowStr) {
  var ss = getSpreadsheet_();
  var lSheet = getOrCreateLockSheet(ss);
  var rows = lSheet.getDataRange().getValues();
  var targetProj = String(projectId || '').trim().toLowerCase();
  var foundRow = -1;
  for (var i = rows.length - 1; i >= 1; i--) {
    if (String(rows[i][0] || '').trim().toLowerCase() === targetProj) {
      foundRow = i + 1;
      break;
    }
  }
  var cleanTier = String(tierCode || 'BASIC_5').trim();
  if (foundRow > 0) {
    lSheet.getRange(foundRow, 10).setValue(cleanTier);
  } else {
    lSheet.appendRow([projectId, customerName || projectId, '', '', 0, nowStr, nowStr, 0, false, cleanTier]);
    try { lSheet.getRange(lSheet.getLastRow(), 9).insertCheckboxes(); } catch (e) {}
  }
  return cleanTier;
}

function handleRequest(body) {
  body = (body && typeof body === 'object') ? body : {};
  var action = String(body.action || (body.deviceId ? 'claim_or_verify' : 'sync_hub')).trim();
  var projectId = String(body.projectId || body.folder || 'wypha').trim();
  var altProjectId = String(body.altProjectId || '').trim();
  var personName = String(body.personName || body.customerName || '').trim();

  // --- 1. PING CLOUD ---
  if (action === 'ping_cloud') {
    setupAllWyphaSheets();

      ok: true,
      version: 'V35-ALL-IN-ONE',
      scriptVersion: 'V35',
      provider: 'gdrive',
      mode: 'GOOGLE_DRIVE_AND_SHEETS_V35_ACTIVE',
      message: 'Koneksi Google Apps Script All-in-One V35 aktif (Menu Reset, Tab Reset Cepat, Broadcast & Drive siap digunakan)!'
    });
  }

  // --- 1B. SET REMOTE TIER DIRECTLY FROM STUDIO ---
  if (action === 'set_remote_tier') {
    var lockTier = LockService.getScriptLock();
    lockTier.waitLock(10000);
    try {
      var updatedTier = upsertRemoteTierForProject(projectId, body.remoteTier || body.defaultRemoteTier || 'BASIC_5', personName, nowStr);
      return jsonResponse({
        ok: true,
        scriptVersion: 'V35',
        projectId: projectId,
        remoteTier: updatedTier,
        message: 'Kolom J di WyphaDeviceLocks berhasil diupdate ke paket: ' + updatedTier
      });
    } finally {
      lockTier.releaseLock();
    }
  }

  // --- 2. UPLOAD FOTO / ASET KE GOOGLE DRIVE ---
  if (action === 'upload_photo') {
    try {
      if (OPTIONAL_UPLOAD_TOKEN && String(body.token || '').trim() !== OPTIONAL_UPLOAD_TOKEN) {
        return jsonResponse({ ok: false, error: 'INVALID_UPLOAD_TOKEN', message: 'Token keamanan upload tidak sesuai.' });
      }
      var dataUrl = String(body.dataUrl || '');
      var match = dataUrl.match(/^data:([^;,]+)?;base64,(.+)$/i);
      if (!match) return jsonResponse({ ok: false, error: 'INVALID_DATA_URL', message: 'Format dataUrl Base64 tidak valid.' });
      var mimeType = match[1] || 'image/webp';
      var bytes = Utilities.base64Decode(match[2]);
      var fileName = String(body.fileName || ('foto-' + new Date().getTime() + '.webp')).trim();
      var folderName = String(body.folder || projectId || 'proyek-wypha').trim();
      var folder = getOrCreateClientFolder(folderName, String(body.parentFolderId || '').trim());
      var file = folder.createFile(Utilities.newBlob(bytes, mimeType, fileName));
      try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
      var fileId = file.getId();
      var thumbUrl = 'https://lh3.googleusercontent.com/d/' + fileId + '=w1600';
      var directViewUrl = 'https://drive.google.com/uc?export=view&id=' + fileId;
      return jsonResponse({
        ok: true,
        provider: 'gdrive',
        fileId: fileId,
        fileName: file.getName(),
        folderName: folder.getName(),
        url: thumbUrl,
        thumbnailUrl: thumbUrl,
        driveViewUrl: directViewUrl
      });
    } catch (err) {
      return jsonResponse({ ok: false, error: String(err.message || err) });
    }
  }

  // --- 3. SYNC HUB (Remote Tier Unlocker + Broadcast + Guestbook + Leaderboard + Milestones) ---
  if (action === 'sync_hub') {
    var ss = getSpreadsheet_();
    var incomingDefaultTier = String(body.defaultRemoteTier || '').trim();
    if (body.forceSyncTier && incomingDefaultTier) {
      upsertRemoteTierForProject(projectId, incomingDefaultTier, personName, nowStr);
    }
    var remoteTier = readRemoteTierForProject(projectId, incomingDefaultTier, altProjectId, personName);
    var broadcast = readActiveBroadcast(projectId, altProjectId, personName);

    var gbSheet = getOrCreateGuestbookSheet(ss);
    var gbRows = gbSheet.getDataRange().getValues();
    var guestbook = [];
    for (var i = gbRows.length - 1; i >= 1 && guestbook.length < 30; i--) {
      if (isTargetProjectMatch_(gbRows[i][1], [projectId, altProjectId]) && String(gbRows[i][6] || 'TAMPIL').trim().toUpperCase() !== 'HAPUS') {
        guestbook.push({
          timestamp: String(gbRows[i][0] || ''),

    }

    var lbSheet = getOrCreateLeaderboardSheet(ss);
    var lbRows = lbSheet.getDataRange().getValues();
    var leaderboard = [];
    for (var j = 1; j < lbRows.length; j++) {
      if (String(lbRows[j][1] || '').trim()) {

      }
    }
    leaderboard.sort(function(a, b) { return (b.totalStars - a.totalStars) || (b.avgScore - a.avgScore); });

    var msSheet = getOrCreateMilestonesSheet(ss);
    var msRows = msSheet.getDataRange().getValues();
    var milestones = [];
    for (var m = msRows.length - 1; m >= 1 && milestones.length < 25; m--) {
      if (isTargetProjectMatch_(msRows[m][1], [projectId, altProjectId])) {

    var chRows = chSheet.getDataRange().getValues();
    var curhat = [];
    for (var k = chRows.length - 1; k >= 1 && curhat.length < 30; k--) {
      if (isTargetProjectMatch_(chRows[k][1], [projectId, altProjectId]) && String(chRows[k][7] || 'TAMPIL').trim().toUpperCase() !== 'HAPUS') {

          sender: String(chRows[k][2] || 'Customer'),

          adminReply: String(chRows[k][5] || ''),
          replyTime: String(chRows[k][6] || '')
        });
      }
    }

    return jsonResponse({
      ok: true,
      scriptVersion: 'V35',
      projectId: projectId,
      remoteTier: remoteTier,
      broadcast: broadcast,
      guestbook: guestbook,
      curhat: curhat,

        projectId: projectId,
        remoteTier: readRemoteTierForProject(projectId, defaultTier, altProjectId, personName),
        broadcast: readActiveBroadcast(projectId, altProjectId, personName)
      });
    }

    var data = sheet.getDataRange().getValues();
    var rowIndex = -1;
    var targetProj = projectId.toLowerCase();
    for (var i = data.length - 1; i >= 1; i--) {
      if (String(data[i][0] || '').trim().toLowerCase() === targetProj) {
        rowIndex = i + 1;
        break;
      }
    }

    if (rowIndex === -1) {
      sheet.appendRow([projectId, customerName || projectId, deviceId, '', 1, nowStr, nowStr, 1, false, defaultTier]);
      try { sheet.getRange(sheet.getLastRow(), 9).insertCheckboxes(); } catch (e) {}
      syncProjectToLegacyAndResetSheets_(projectId, customerName || projectId, [deviceId], 'TERKUNCI (1/2)', nowStr, defaultTier, 'Klaim Slot 1/2 (HP ke-1 Aktif)');
      return jsonResponse({
        ok: true,
        scriptVersion: 'V35',
        status: 'CLAIMED_SLOT_1_OF_2',
        slot: 1,
        maxDevices: MAX_DEVICES,
        usedSlots: 1,
        ownerDeviceIds: [deviceId],
        ownerDeviceId: deviceId,
        remoteTier: defaultTier,
        broadcast: readActiveBroadcast(projectId, altProjectId, personName)
      });
    }

    var rowVals = sheet.getRange(rowIndex, 1, 1, 10).getValues()[0];
    var resetCmd = String(rowVals[8] || '').trim().toUpperCase();
    var remoteTierVal = String(rowVals[9] || '').trim();
    if (!remoteTierVal && defaultTier) {
      remoteTierVal = defaultTier;
      sheet.getRange(rowIndex, 10).setValue(defaultTier);
    }
    var dev1 = String(rowVals[2] || '').trim().toUpperCase();
    var dev2 = String(rowVals[3] || '').trim().toUpperCase();

    if (resetCmd === 'RESET' || resetCmd === 'YA' || resetCmd === '1' || resetCmd === 'TRUE') {
      dev1 = '';
      dev2 = '';
      sheet.getRange(rowIndex, 9).setValue(false);
    }

    var owners = [dev1, dev2].filter(Boolean);
    var visits = (Number(rowVals[7]) || 0) + 1;
    if (customerName && !String(rowVals[1] || '').trim()) {
      sheet.getRange(rowIndex, 2).setValue(customerName);
    }

    if (owners.indexOf(deviceId) !== -1) {
      sheet.getRange(rowIndex, 7, 1, 2).setValues([[nowStr, visits]]);
      syncProjectToLegacyAndResetSheets_(projectId, customerName || String(rowVals[1] || projectId), owners, owners.length >= 2 ? 'TERKUNCI PENUH (2/2)' : 'TERKUNCI (1/2)', nowStr, remoteTierVal, 'Verifikasi Ulang HP ke-' + (owners.indexOf(deviceId) + 1));
      return jsonResponse({
        ok: true,
        scriptVersion: 'V35',
        status: 'VERIFIED_OWNER_DEVICE',
        slot: owners.indexOf(deviceId) + 1,
        maxDevices: MAX_DEVICES,
        usedSlots: owners.length,
        ownerDeviceIds: owners,
        ownerDeviceId: owners.join(' | '),
        remoteTier: remoteTierVal,
        broadcast: readActiveBroadcast(projectId, altProjectId, personName)
      });
    }

    if (owners.length < MAX_DEVICES) {
      owners.push(deviceId);
      sheet.getRange(rowIndex, 3, 1, 3).setValues([[owners[0] || '', owners[1] || '', owners.length]]);
      sheet.getRange(rowIndex, 7, 1, 2).setValues([[nowStr, visits]]);
      syncProjectToLegacyAndResetSheets_(projectId, customerName || String(rowVals[1] || projectId), owners, owners.length >= 2 ? 'TERKUNCI PENUH (2/2)' : 'TERKUNCI (1/2)', nowStr, remoteTierVal, owners.length === 1 ? 'Klaim Slot 1/2' : 'Klaim Slot 2/2');
      return jsonResponse({
        ok: true,
        scriptVersion: 'V35',
        status: owners.length === 1 ? 'CLAIMED_SLOT_1_OF_2' : 'CLAIMED_SLOT_2_OF_2',
        slot: owners.length,
        maxDevices: MAX_DEVICES,
        usedSlots: owners.length,
        ownerDeviceIds: owners,
        ownerDeviceId: owners.join(' | '),
        remoteTier: remoteTierVal,
        broadcast: readActiveBroadcast(projectId, altProjectId, personName)
      });
    }

    syncProjectToLegacyAndResetSheets_(projectId, customerName || String(rowVals[1] || projectId), owners, 'TERKUNCI PENUH (2/2)', nowStr, remoteTierVal, 'HP ke-3+ Ditolak (' + deviceId + ') pada ' + nowStr);
    return jsonResponse({
      ok: false,
      scriptVersion: 'V35',
      status: 'REJECTED_MAX_2_DEVICES',
      reason: 'MAX_2_DEVICES_ALREADY_CLAIMED',
      maxDevices: MAX_DEVICES,
      usedSlots: owners.length,
      ownerDeviceIds: owners,
      ownerDeviceId: owners.join(' | '),
      remoteTier: remoteTierVal,
      broadcast: readActiveBroadcast(projectId, altProjectId, personName)
    });
  } finally {
    lock.releaseLock();
  }
}
